// ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
// Example file: rename it to .cs to enable it.
// It lets you tune GreaterRiftBossTimerPlugin without editing the plugin itself.
//
// NOTE: this HUD install disables every plugin its manager does not know about.
// If you enable this file, ALSO add it to that manager's enable list:
//   "Turbo.Plugins.GuiSquare.GreaterRiftBossTimerCustomizer"
// ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^

using Turbo.Plugins.Default;

namespace Turbo.Plugins.GuiSquare
{
    public class GreaterRiftBossTimerCustomizer : BasePlugin, ICustomizer
    {
        public GreaterRiftBossTimerCustomizer()
        {
            Enabled = true;
        }

        public void Customize()
        {
            Hud.RunOnPlugin<GreaterRiftBossTimerPlugin>(plugin =>
            {
                // ---------------------------------------------------------------
                // 1) Where the block sits
                // ---------------------------------------------------------------
                // BelowMinimapText -> minimap top-left corner, PanelLineCount lines down
                // Custom           -> free placement through CustomX / CustomY
                plugin.Anchor = GrBossTimerAnchor.BelowMinimapText;

                // How far below the minimap's top edge the block starts, counted in text
                // lines. THIS is the value to adjust if the block overlaps the text
                // above it or floats too low: count the lines already drawn over the
                // minimap and land right under the last one.
                //
                // The line height is measured live, including the 13.5px floor of that
                // text, so the block keeps its place when the game window is resized.
                plugin.PanelLineCount = 8;

                // Whatever sits below these three lines has to move down by three lines
                // to make room, through whatever setting places it.

                // Fine tuning, as a ratio of the screen height.
                // plugin.OffsetX = 0.000f;
                // plugin.OffsetY = 0.000f;

                // Extra air between lines, as a fraction of a line height.
                // plugin.LineSpacing = 0.0f;

                // Fully free placement:
                // plugin.Anchor = GrBossTimerAnchor.Custom;
                // plugin.CustomX = 0.02f;
                // plugin.CustomY = 0.30f;

                // ---------------------------------------------------------------
                // 2) What the three lines say
                // ---------------------------------------------------------------
                // plugin.CurrentLabel = "Bosskill: ";
                // plugin.AverageLabel = "Bossavg: ";
                // plugin.SessionLabel = "Session: ";
                // plugin.IdlePlaceholder = "-";

                // " x12" behind the average: how many kills it is built on.
                // plugin.ShowSampleCount = true;

                // Third line, the fastest and slowest kill of the session, as
                //   Session: best 8.4s // worst 41.2s
                // On by default. The kill count and the average are left off
                // it, since the line above already carries both.
                // plugin.ShowSessionLine = true;

                // Put the guardian's name on the current line.
                // plugin.ShowGuardianName = false;

                // Draw nothing at all until the first guardian of the session is engaged.
                // plugin.HideUntilFirstBoss = false;

                // plugin.HideOnMapModes = true;
                // plugin.HideInTown = false;

                // ---------------------------------------------------------------
                // 3) What exactly is being measured
                // ---------------------------------------------------------------
                // BecomesAttackable -> the clock starts when the guardian can first be
                //                      hit, so its spawn animation -- untargetable,
                //                      immune, and the same length every time -- is
                //                      left out of the measurement            [default]
                // FirstDamage       -> the clock starts on the guardian's first point of
                //                      damage, which also leaves out the time spent
                //                      closing in or setting up
                // GuardianSpawn     -> the clock starts the moment the guardian spawns,
                //                      spawn animation included
                // plugin.StartOn = GrBossTimerStart.BecomesAttackable;

                // true  -> game ticks (60 per second), what the rift timer itself counts,
                //          and the one measure a changed game speed cannot skew [default]
                // false -> real elapsed time on the wall clock
                // plugin.UseGameTime = true;

                // Fraction of max health the guardian has to drop below to count as
                // damaged. Used by StartOn = FirstDamage, and as the backstop of
                // BecomesAttackable: a guardian taking damage is attackable whatever its
                // flags claim, which keeps one whose flags never clear from reporting a
                // fight of no duration.
                // plugin.FirstDamageThreshold = 0.999d;

                // How long a guardian stays unhittable after the progress bar fills.
                // Measured, not guessed: over 220 kills across 25 distinct guardians the
                // gap landed in 222..227 ticks 215 times, so it is a fixed game rule
                // rather than a per-guardian animation length.
                //
                // The flags stay the primary signal -- they self-adjust if the game ever
                // changes this. The value only polices them: when the observed transition
                // sits further than the tolerance from it, the observation is treated as
                // no observation at all and replaced.
                //
                // The tolerance sits in a gap the data leaves wide open. Over 220 kills
                // the sound readings never strayed more than 3 ticks, and the nearest bad
                // one was 31 out -- so 15 is five times the jitter and half the closest
                // fault. It catches all 5 bad readings and leaves the other 215 alone.
                // plugin.SpawnProtectionTicks = 225;
                // plugin.SpawnProtectionToleranceTicks = 15;

                // Kills shorter than this are still shown, but left out of the average.
                // 0 keeps every sample: a one-shot guardian in a low rift really does die
                // in no time.
                // plugin.MinimumValidMilliseconds = 0d;

                // How long to wait, in real milliseconds, before ruling on a guardian
                // that has dropped off the actor list. A kill is only counted if the rift
                // has actually moved on to its reward step, so a hero respawning far from
                // the fight never invents one.
                // plugin.VanishGraceMs = 900;

                // Wipe the session statistics on every new game. false keeps them for as
                // long as TurboHUD runs.
                // plugin.ResetStatsOnNewGame = false;

                // ---------------------------------------------------------------
                // 4) Look
                // ---------------------------------------------------------------
                // The font size is re-derived from the window, the same way the tracker
                // text above derives its own, so the two match at any resolution. That
                // text is not drawn at a fixed size: it grows until it fills the minimap
                // width, stopping early only once a line is at least MinLineHeight tall
                // AND the size has reached MaxFontSize. TurboHUD sizes scale with the
                // window, so at 800x600 MaxFontSize gives lines well under the floor,
                // the cap is not met, and both keep growing past it together.
                // plugin.AutoFontSize = true;
                // plugin.MaxFontSize = 8.0f;     // the tracker text's own cap
                // plugin.MinLineHeight = 13.5f;  // and its own absolute floor, in pixels

                // Turning the auto-sizing off hands the fonts back to you, drawn exactly
                // as built here. Arial bold, no shadow, the running one tinted while the
                // fight is still on.
                // plugin.AutoFontSize = false;
                // plugin.TextFont    = Hud.Render.CreateFont("Arial", 8.0f, 255, 255, 255, 255, true, false, false);
                // plugin.RunningFont = Hud.Render.CreateFont("Arial", 8.0f, 255, 255, 225, 130, true, false, false);

                // ---------------------------------------------------------------
                // 5) Diagnostics
                // ---------------------------------------------------------------
                // One line per rift appended to a log file. Independent of the on-screen
                // panel: over a long run the panel only keeps the last few rifts, the
                // log keeps them all and can be read back afterwards. Off by default: turn
                // it on to record a run worth investigating.
                //
                // The path is relative to the TurboHUD folder and deliberately NOT under
                // plugins/ -- writing there trips the file watcher and forces a recompile
                // on every rift.
                // plugin.LogToFile = false;
                // plugin.LogFileName = "GuiSquare/gr_boss_timer_log.txt";

                // Names the exact state the tracker is in, the guardian it locked onto,
                // the numbers behind every line, and one entry per rift that has ended.
                // plugin.DebugEnabled = true;
                // plugin.DebugX = 0.35f;
                // plugin.DebugY = 0.20f;
            });
        }
    }
}
